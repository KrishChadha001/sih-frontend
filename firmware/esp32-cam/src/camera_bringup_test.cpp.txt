// CIRCUIT BRING-UP TEST - not part of the real firmware.
//
// Confirms the physical camera connection works before adding WiFi/server
// complexity: no network, no credentials needed. Just power, watch serial.
//
// HOW TO USE:
//   1. Rename src/main.cpp to src/main.cpp.bak (keep it - that's the real
//      firmware, you'll swap back to it after this test passes).
//   2. Rename this file to src/main.cpp.
//   3. pio run --target upload  (GPIO0 tied to GND, as usual)
//   4. Remove the GPIO0-GND jumper, press reset.
//   5. pio device monitor
//
// WHAT TO EXPECT: every 3 seconds, either:
//   "[OK] captured frame, N bytes"          <- camera + ribbon cable are
//                                               genuinely working
//   "[FAIL] camera init failed: 0x..."      <- see camera_pins.h /
//                                               README's Troubleshooting
//                                               section (almost always
//                                               ribbon seating or power)
//   "[FAIL] capture returned null"          <- initialized but couldn't
//                                               grab a frame - still a
//                                               camera/power problem

#include <Arduino.h>
#include <esp_camera.h>
#include "camera_pins.h"

namespace {

bool initCamera() {
    camera_config_t config;
    config.ledc_channel = LEDC_CHANNEL_0;
    config.ledc_timer = LEDC_TIMER_0;
    config.pin_d0 = Y2_GPIO_NUM;
    config.pin_d1 = Y3_GPIO_NUM;
    config.pin_d2 = Y4_GPIO_NUM;
    config.pin_d3 = Y5_GPIO_NUM;
    config.pin_d4 = Y6_GPIO_NUM;
    config.pin_d5 = Y7_GPIO_NUM;
    config.pin_d6 = Y8_GPIO_NUM;
    config.pin_d7 = Y9_GPIO_NUM;
    config.pin_xclk = XCLK_GPIO_NUM;
    config.pin_pclk = PCLK_GPIO_NUM;
    config.pin_vsync = VSYNC_GPIO_NUM;
    config.pin_href = HREF_GPIO_NUM;
    config.pin_sccb_sda = SIOD_GPIO_NUM;
    config.pin_sccb_scl = SIOC_GPIO_NUM;
    config.pin_pwdn = PWDN_GPIO_NUM;
    config.pin_reset = RESET_GPIO_NUM;
    config.xclk_freq_hz = 20000000;
    config.pixel_format = PIXFORMAT_JPEG;

    bool hasPsram = psramFound();
    Serial.printf("[Camera] PSRAM found: %s\n", hasPsram ? "yes" : "NO");

    if (hasPsram) {
        config.frame_size = FRAMESIZE_VGA;
        config.jpeg_quality = 12;
        config.fb_count = 2;
        config.fb_location = CAMERA_FB_IN_PSRAM;
        config.grab_mode = CAMERA_GRAB_LATEST;
    } else {
        config.frame_size = FRAMESIZE_SVGA;
        config.jpeg_quality = 15;
        config.fb_count = 1;
        config.fb_location = CAMERA_FB_IN_DRAM;
    }

    esp_err_t err = esp_camera_init(&config);
    if (err != ESP_OK) {
        Serial.printf("[FAIL] camera init failed: 0x%x\n", err);
        return false;
    }
    return true;
}

} // namespace

void setup() {
    Serial.begin(115200);
    delay(500);
    Serial.println("\n[Bring-up test] booting...");

    if (!initCamera()) {
        Serial.println("[Bring-up test] halting - fix the camera connection and re-flash");
        while (true) {
            delay(1000);
        }
    }
    Serial.println("[Bring-up test] camera initialized OK, starting capture loop");
}

void loop() {
    camera_fb_t *fb = esp_camera_fb_get();
    if (fb == nullptr) {
        Serial.println("[FAIL] capture returned null");
    } else {
        Serial.printf("[OK] captured frame, %u bytes (%dx%d)\n", fb->len, fb->width, fb->height);
        esp_camera_fb_return(fb);
    }
    delay(3000);
}
